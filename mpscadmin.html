Supabase SQL Setup Script (Run in Supabase SQL Editor)

Open your Supabase Project Dashboard -> SQL Editor -> New Query, paste the SQL below and click RUN: 

-- 1. Create events table
create table public.events (
  id uuid default gen_random_uuid() primary key,
  day text not null,
  time text not null,
  title text not null,
  venue text not null,
  description text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 2. Create ec_members table
create table public.ec_members (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  role text not null,
  wing text not null,
  image_url text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 3. Enable Public Read & Insert Policies
alter table public.events enable row level security;
alter table public.ec_members enable row level security;

create policy "Public Access Events" on public.events for all using (true);
create policy "Public Access EC" on public.ec_members for all using (true);
