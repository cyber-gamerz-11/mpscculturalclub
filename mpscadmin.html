Supabase SQL Setup Script (Run in Supabase SQL Editor)

Open your Supabase Project Dashboard -> SQL Editor -> New Query, paste the SQL below and click RUN: 

-- 1. Create schedule events table
create table if not exists public.events (
  id uuid default gen_random_uuid() primary key,
  day text not null,
  time text not null,
  title text not null,
  venue text not null,
  description text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 2. Create ec_members table
create table if not exists public.ec_members (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  role text not null,
  wing text not null default 'BVB',
  image_url text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 3. Add wing & image_url columns, and remove NOT NULL on legacy batch column if it exists
alter table public.ec_members add column if not exists wing text default 'BVB';
alter table public.ec_members add column if not exists image_url text;
do $$ 
begin
  if exists (select 1 from information_schema.columns where table_name='ec_members' and column_name='batch') then
    alter table public.ec_members alter column batch drop not null;
  end if;
end $$;

-- 4. Create segments table
create table if not exists public.segments (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  tag text not null default 'Cultural Segment',
  icon text not null default 'fa-star',
  description text,
  day_info text default 'Festival Day',
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 5. Create segment_events table
create table if not exists public.segment_events (
  id uuid default gen_random_uuid() primary key,
  segment_id text not null,
  title text not null,
  description text,
  time text,
  venue text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 6. Create event_groups table
create table if not exists public.event_groups (
  id uuid default gen_random_uuid() primary key,
  event_id text not null,
  group_name text not null,
  age_limit text,
  rules text,
  description text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 7. Enable RLS and Create Full Access Policies
alter table public.events enable row level security;
alter table public.ec_members enable row level security;
alter table public.segments enable row level security;
alter table public.segment_events enable row level security;
alter table public.event_groups enable row level security;

drop policy if exists "Public Access Events" on public.events;
drop policy if exists "Public Access EC" on public.ec_members;
drop policy if exists "Public Access Segments" on public.segments;
drop policy if exists "Public Access Segment Events" on public.segment_events;
drop policy if exists "Public Access Event Groups" on public.event_groups;

create policy "Public Access Events" on public.events for all using (true) with check (true);
create policy "Public Access EC" on public.ec_members for all using (true) with check (true);
create policy "Public Access Segments" on public.segments for all using (true) with check (true);
create policy "Public Access Segment Events" on public.segment_events for all using (true) with check (true);
create policy "Public Access Event Groups" on public.event_groups for all using (true) with check (true);
